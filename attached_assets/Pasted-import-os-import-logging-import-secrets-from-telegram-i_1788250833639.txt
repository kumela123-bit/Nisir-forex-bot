async def button_handler(
update: Update,
context: ContextTypes.DEFAULT_TYPE,
):

query = update.callback_query
await query.answer()
action = query.data
user_id = query.from_user.id
# =================================
# MAIN MENU
# =================================
if action == "main_menu":
    await query.edit_message_text(
        "Welcome to Nisir Forex Academy.\n\n"
        "Choose a service below to get started.",
        reply_markup=main_menu(user_id),
    )
    return
# =================================
# COURSES
# =================================
if action == "courses":
    await query.edit_message_text(
        "📚 COURSE\n\n"
        "Choose your course below.",
        reply_markup=courses_menu(),
    )
    return
# =================================
# BASIC SMC
# =================================
if action == "basic_smc":
    text = (
        "📚 Basic Smart Money Concept\n"
        "For Beginners\n\n"
        "✅ Market Structure\n"
        "✅ Liquidity Concepts\n"
        "✅ Entry & Exit Basics\n"
        "✅ Risk Management\n"
        "❌ Personalized Trading Plan\n"
        "❌ Lifetime Support\n"
        "❌ Account flips: $10→$100 | $100→$1,000\n\n"
        "💰 $39"
    )
    await query.edit_message_text(
        text,
        reply_markup=buy_now_menu(
            "basic_smc",
            "courses",
        ),
    )
    return
# =================================
# ADVANCED SMC
# =================================
if action == "advanced_smc":
    text = (
        "🎓 Advanced SMC Mastery\n"
        "For Serious Traders\n\n"
        "✅ Institutional Direction Analysis\n"
        "✅ Liquidity Behavior\n"
        "✅ Advanced Trade Management\n"
        "✅ Psychology & Discipline\n"
        "❌ Personalized Trading Plan\n"
        "❌ Lifetime Support\n"
        "❌ Account flips: $10→$100 | $100→$1,000\n\n"
        "💰 $59"
    )
    await query.edit_message_text(
        text,
        reply_markup=buy_now_menu(
            "advanced_smc",
            "courses",
        ),
    )
    return
# =================================
# MENTORSHIP
# =================================
if action == "mentorship":
    text = (
        "🤝 Private 1-to-1 Mentorship\n"
        "Limited Slots Available\n\n"
        "✅Personalized Trading Plan\n"
        "✅Live Chart Reviews\n"
        "✅Account Growth Guidance\n"
        "✅Lifetime Support\n"
        "✅Live Q&A sessions\n"
        "✅Account flips: $10→$100 | $100→$1,000\n"
        "✅Certificate of completion\n\n"
        "💰 $99"
    )
    await query.edit_message_text(
        text,
        reply_markup=buy_now_menu(
            "mentorship",
            "main_menu",
        ),
    )
    return
# =================================
# VIP SIGNAL
# =================================
if action == "vip_signal":
    text = (
        "📈 VIP Signal\n"
        "For Serious Traders Who Want Real Results\n\n"
        "✅ High-Probability Signals\n"
        "✅ Smart Money Technique Based\n"
        "✅ Daily Trade Alerts\n"
        "✅ Risk & Money Management Included\n\n"
        "💰 $20"
    )
    await query.edit_message_text(
        text,
        reply_markup=buy_now_menu(
            "vip",
            "main_menu",
        ),
    )
    return
# =================================
# BUY NOW
# =================================
if action.startswith("buy_"):
    product_id = action.replace("buy_", "", 1)
    product = PRODUCTS.get(product_id)
    if not product:
        await query.answer(
            "Product not found.",
            show_alert=True,
        )
        return
    payments[user_id] = {
        "user_id": user_id,
        "product_id": product_id,
        "product_name": product["name"],
        "price": product["price"],
        "status": "pending",
        "receipt_file_id": None,
    }
    user_states[user_id] = "payment_method"
    await query.edit_message_text(
        "💳 Payment\n\n"
        f"Product: {product['name']}\n"
        f"Amount: {product['price']}\n\n"
        "Choose your payment method:",
        reply_markup=payment_method_menu(),
    )
    return
# =================================
# BANK TRANSFER
# =================================
if action == "bank_transfer":
    if user_id not in payments:
        await query.answer(
            "Please select a product first.",
            show_alert=True,
        )
        return
    payment = payments[user_id]
    user_states[user_id] = "waiting_i_paid"
    text = (
        "⭐ PAYMENT METHOD\n\n"
        f"Product: {payment['product_name']}\n"
        f"Amount: {payment['price']}\n\n"
        "Account Holder Name:\n"
        "Kidus Efrem\n\n"
        "CBE:\n"
        "1000732883518\n\n"
        "Abyssinia Bank:\n"
        "153963878\n\n"
        "Telebirr:\n"
        "0976432160\n\n"
        "After making the payment, click "
        "the button below."
    )
    await query.edit_message_text(
        text,
        reply_markup=i_paid_menu(),
    )
    return
# =================================
# I PAID
# =================================
if action == "i_paid":
    if user_id not in payments:
        await query.answer(
            "Payment session not found.",
            show_alert=True,
        )
        return
    user_states[user_id] = "waiting_receipt"
    await query.edit_message_text(
        "📸 Please send a screenshot/photo "
        "of your payment receipt here."
    )
    return
# =================================
# SUBMIT PAYMENT
# =================================
if action == "submit_payment":
    if user_id not in payments:
        await query.answer(
            "Payment not found.",
            show_alert=True,
        )
        return
    payment = payments[user_id]
    if not payment.get("receipt_file_id"):
        await query.answer(
            "Please upload your payment receipt first.",
            show_alert=True,
        )
        return
    payment["status"] = "under_review"
    # Send payment to admin
    try:
        await context.bot.send_message(
            chat_id=ADMIN_TELEGRAM_ID,
            text=(
                "🔔 NEW PAYMENT SUBMISSION\n\n"
                f"👤 User ID: {user_id}\n"
                f"📦 Product: {payment['product_name']}\n"
                f"💰 Amount: {payment['price']}\n\n"
                "⏳ Status: UNDER REVIEW"
            ),
        )
        await context.bot.send_photo(
            chat_id=ADMIN_TELEGRAM_ID,
            photo=payment["receipt_file_id"],
            caption=(
                f"📸 Payment Receipt\n\n"
                f"👤 User ID: {user_id}\n"
                f"📦 {payment['product_name']}\n"
                f"💰 {payment['price']}"
            ),
            reply_markup=admin_payment_menu(user_id),
        )
    except Exception:
        logger.exception(
            "Failed to send payment to admin."
        )
    await query.edit_message_text(
        "✅ Payment receipt received!\n\n"
        "Thank you. Your payment is now under review. 🔍\n\n"
        "⏳ Please wait while our team verifies "
        "your payment.\n\n"
        "You will be notified once your payment "
        "is approved. 💳✅",
        reply_markup=main_menu_button(user_id),
    )
    user_states[user_id] = None
    return
# =================================
# ADMIN DASHBOARD
# =================================
if action == "admin_dashboard":
    if user_id != ADMIN_TELEGRAM_ID:
        await query.answer(
            "Access denied.",
            show_alert=True,
        )
        return
    await query.edit_message_text(
        "⚙️ ADMIN DASHBOARD\n\n"
        "Payment submissions are sent here "
        "for manual review.\n\n"
        "Approve or reject each payment "
        "after checking the receipt.",
        reply_markup=main_menu_button(user_id),
    )
    return
# =================================
# ADMIN APPROVE
# =================================
if action.startswith("approve_"):
    if user_id != ADMIN_TELEGRAM_ID:
        await query.answer(
            "Access denied.",
            show_alert=True,
        )
        return
    target_user_id = int(
        action.replace("approve_", "", 1)
    )
    payment = payments.get(target_user_id)
    if not payment:
        await query.answer(
            "Payment record not found.",
            show_alert=True,
        )
        return
    payment["status"] = "approved"
    # VIP Signal approval
    if payment["product_id"] == "vip":
        try:
            invite_link = await context.bot.create_chat_invite_link(
                chat_id=VIP_CHANNEL_ID,
                member_limit=1,
            )
            await context.bot.send_message(
                chat_id=target_user_id,
                text=(
                    "✅ Payment approved.\n\n"
                    "Your VIP invite link is ready:\n"
                    f"{invite_link.invite_link}"
                ),
            )
            payment["vip_invite_link"] = (
                invite_link.invite_link
            )
        except Exception:
            logger.exception(
                "Failed to create VIP invite link."
            )
            await context.bot.send_message(
                chat_id=target_user_id,
                text=(
                    "✅ Payment approved.\n\n"
                    "Your VIP access is approved, "
                    "but the invite link could not be "
                    "generated automatically. "
                    "Please contact support."
                ),
            )
    else:
        await context.bot.send_message(
            chat_id=target_user_id,
            text=(
                "✅ Payment approved.\n\n"
                "Your purchase has been approved."
            ),
        )
    await query.edit_message_reply_markup(
        reply_markup=None
    )
    await query.message.reply_text(
        "✅ Payment approved successfully."
    )
    return
# =================================
# ADMIN REJECT
# =================================
if action.startswith("reject_"):
    if user_id != ADMIN_TELEGRAM_ID:
        await query.answer(
            "Access denied.",
            show_alert=True,
        )
        return
    target_user_id = int(
        action.replace("reject_", "", 1)
    )
    payment = payments.get(target_user_id)
    if not payment:
        await query.answer(
            "Payment record not found.",
            show_alert=True,
        )
        return
    payment["status"] = "rejected"
    await context.bot.send_message(
        chat_id=target_user_id,
        text=(
            "❌ Payment rejected.\n\n"
            "Your payment could not be verified.\n"
            "Please contact support if you believe "
            "this was a mistake."
        ),
    )
    await query.edit_message_reply_markup(
        reply_markup=None
    )
    await query.message.reply_text(
        "❌ Payment rejected."
    )
    return
# =================================
# MY ACCOUNT
# =================================
if action == "my_account":
    await query.edit_message_text(
        "👤 My Account\n\n"
        f"Telegram ID: {user_id}\n\n"
        "Your account information will appear here.",
        reply_markup=main_menu_button(user_id),
    )
    return
# =================================
# PAYMENT STATUS
# =================================
if action == "payment_status":
    payment = payments.get(user_id)
    if not payment:
        text = (
            "💳 Payment Status\n\n"
            "No payment submission found."
        )
    else:
        status = payment["status"]
        text = (
            "💳 Payment Status\n\n"
            f"📦 Product: {payment['product_name']}\n"
            f"💰 Amount: {payment['price']}\n"
            f"📌 Status: {status.replace('_', ' ').title()}"
        )
    await query.edit_message_text(
        text,
        reply_markup=main_menu_button(user_id),
    )
    return

async def photo_handler(
update: Update,
context: ContextTypes.DEFAULT_TYPE,
):

user_id = update.effective_user.id
if user_states.get(user_id) != "waiting_receipt":
    return
if user_id not in payments:
    return
payment = payments[user_id]
photo = update.message.photo[-1]
payment["receipt_file_id"] = photo.file_id
user_states[user_id] = "receipt_received"
await update.message.reply_text(
    "✅ Payment receipt received!\n\n"
    "Thank you. Your payment is now under review. 🔍\n\n"
    "⏳ Please wait while our team verifies "
    "your payment.\n\n"
    "You will be notified once your payment "
    "is approved. 💳✅"
)
# Automatically send to admin for review
try:
    await context.bot.send_message(
        chat_id=ADMIN_TELEGRAM_ID,
        text=(
            "🔔 NEW PAYMENT SUBMISSION\n\n"
            f"👤 User ID: {user_id}\n"
            f"📦 Product: {payment['product_name']}\n"
            f"💰 Amount: {payment['price']}\n\n"
            "⏳ Status: UNDER REVIEW"
        ),
    )
    await context.bot.send_photo(
        chat_id=ADMIN_TELEGRAM_ID,
        photo=photo.file_id,
        caption=(
            "📸 PAYMENT RECEIPT\n\n"
            f"👤 User ID: {user_id}\n"
            f"📦 Product: {payment['product_name']}\n"
            f"💰 Amount: {payment['price']}\n\n"
            "Please manually verify this payment."
        ),
        reply_markup=admin_payment_menu(user_id),
    )
except Exception:
    logger.exception(
        "Failed to send receipt to admin."
    )

def main():

application = (
    Application.builder()
    .token(TOKEN)
    .build()
)

    CommandHandler("start", start)
)

)

        filters.PHOTO,
        photo_handler,
    )
)
application.add_error_handler(
    error_handler
)
logger.info(
    "Nisir Forex Academy Bot is starting..."
)
application.run_polling()

if name == “main”:
main()
